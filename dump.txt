import Navbar from './components/Navbar.jsx';
import Hero from './components/Hero.jsx';
import { lazy, Suspense, useEffect } from 'react';

const MenuSection = lazy(() => import('./components/MenuSection.jsx'));
const GuestLove = lazy(() => import('./components/GuestLove.jsx'));
const CourtyardVibe = lazy(() => import('./components/CourtyardVibe.jsx'));
const VisitUs = lazy(() => import('./components/VisitUs.jsx'));
const Footer = lazy(() => import('./components/Footer.jsx'));

const BelowFoldFallback = () => <div aria-hidden="true" style={{ minHeight: '420px' }} />;

const App = () => {
 useEffect(() => {
  if (window.location.hash) {
   window.history.replaceState(null, '', window.location.pathname + window.location.search);
  }
 }, []);
 return (
	<main>
   <div className="grain-overlay" aria-hidden="true" />
   <div className="court-yard-glow glow-amber" aria-hidden="true" />
   <div className="court-yard-glow glow-red" aria-hidden="true" />
   <div className="court-yard-glow glow-green" aria-hidden="true" />
	 <Navbar />
	 <Hero />
   <Suspense fallback={<BelowFoldFallback />}><MenuSection /></Suspense>
   <Suspense fallback={<BelowFoldFallback />}><GuestLove /></Suspense>
   <Suspense fallback={<BelowFoldFallback />}><CourtyardVibe /></Suspense>
   <Suspense fallback={<BelowFoldFallback />}><VisitUs /></Suspense>
   <Suspense fallback={<div aria-hidden="true" style={{ minHeight: '430px' }} />}><Footer /></Suspense>
	</main>
 )
}

export default App

import { useEffect, useState } from 'react';
import { navLinks } from '../constants/index.js';
import { scrollToSection } from '../utils/navigation.js';

const Navbar = () => {
  const [open, setOpen] = useState(false);
  const [scrolled, setScrolled] = useState(false);
  const [active, setActive] = useState('hero');

  useEffect(() => {
    const onScroll = () => setScrolled(window.scrollY > 24);
    onScroll();
    window.addEventListener('scroll', onScroll, { passive: true });
    const observer = new IntersectionObserver((entries) => {
      entries.forEach((entry) => {
        if (entry.isIntersecting) setActive(entry.target.id);
      });
    }, { rootMargin: '-35% 0px -55%', threshold: 0 });
    ['hero', ...navLinks.map((link) => link.id)].forEach((id) => {
      const element = document.getElementById(id);
      if (element) observer.observe(element);
    });
    return () => {
      window.removeEventListener('scroll', onScroll);
      observer.disconnect();
    };
  }, []);

  const navigate = (event, id) => {
    event.preventDefault();
    setOpen(false);
    scrollToSection(id);
  };
  return (
    <header className={scrolled ? 'site-nav scrolled' : 'site-nav'}>
      <button className="brand" type="button" onClick={(event) => navigate(event, 'hero')} aria-label="Trini-Yaad home">
        <img className="navbar-logo" src="/images/logo-clean.webp?v=2" width="240" height="96" fetchPriority="high" loading="eager" decoding="async" alt="Trini-Yaad at Dubwise Cafe" />
      </button>
      <button className="menu-toggle" type="button" aria-expanded={open} onClick={() => setOpen(!open)}>
        <span /><span />
      </button>
      <nav className={open ? 'nav-links open' : 'nav-links'} aria-label="Main navigation">
        {navLinks.map((link) => (
          <button key={link.id} className={active === link.id ? 'active' : ''} type="button" onClick={(event) => navigate(event, link.id)}>{link.label}<span /></button>
        ))}
        <button className="nav-order" type="button" onClick={(event) => navigate(event, 'menu')}>Order food</button>
      </nav>
    </header>
  );
};
export default Navbar;
